from __future__ import annotations

import json
import sys
from pathlib import Path

import textops
import numops


def main(argv):
    here = Path(__file__).resolve().parent
    spec = json.loads((here.parent / "spec.json").read_text())
    payload = json.loads(Path(argv[1]).read_text())
    out = {"texts": textops.transform(spec["text_op"], payload["texts"]),
           "numbers": numops.summarize(spec["num_op"], payload["numbers"])}
    Path(argv[2]).write_text(json.dumps(out))
    return 0


if __name__ == "__main__":
    raise SystemExit(main(sys.argv))

--- src/numops.py ---
def summarize(op, numbers):
    if op == "sum":
        return sum(numbers, 0)
    if op == "span":
        return (max(numbers) - min(numbers)) if numbers else 0
    if op == "rmean":
        return round(sum(numbers, 0) / len(numbers), 2) if numbers else 0
    if op == "count_pos":
        return sum(1 for n in numbers if n > 0)
    raise ValueError(op)

def transform(op, texts):
    if op == "strip_lower":
        return [t.strip().lower() for t in texts]
    if op == "dedup_sort":
        return sorted(set(texts))
    if op == "reverse":
        return [t[::-1] for t in texts]
    if op == "upper":
        return [t.upper() for t in texts]
    raise ValueError(op)
